```python
# 数据结构名称：LRU Cache (Least Recently Used Cache)
# 原理说明：
# LRU Cache是一种缓存淘汰策略，当缓存空间已满时，会优先淘汰最长时间未被使用的（Least Recently Used）数据项。
#
# 实现LRU Cache通常需要结合两种数据结构：
# 1.  哈希表 (Python中的字典 dict)：用于存储键到数据项的映射。这使得我们能够以O(1)的平均时间复杂度快速查找、插入和更新数据项。
#     哈希表存储的是 `key -> Node` 的映射，其中 `Node` 是双向链表中的节点。
# 2.  双向链表 (Doubly Linked List)：用于维护数据项的使用顺序。
#     链表头（或靠近头的一端）存储最新使用的数据项 (Most Recently Used, MRU)，链表尾（或靠近尾的一端）存储最长时间未使用的 (Least Recently Used, LRU) 数据项。
#     当一个数据项被访问（get）或更新（put）时，它会被移动到链表的MRU端。
#     当缓存空间不足时，链表尾部的LRU数据项会被移除。
#
# 这种组合使得所有核心操作（get, put）都能在O(1)的平均时间复杂度内完成。

# 定义双向链表节点
class Node:
    """
    双向链表的节点。
    每个节点存储键、值，以及指向前一个和后一个节点的指针。
    键是必需的，因为在淘汰节点时，我们需要从哈希表中删除对应的条目。
    """
    def __init__(self, key, value):
        self.key = key
        self.value = value
        self.prev = None  # 指向前一个节点
        self.next = None  # 指向后一个节点

class LRUCache:
    """
    LRU Cache的实现。
    """
    def __init__(self, capacity: int):
        """
        初始化LRU Cache。
        :param capacity: 缓存的最大容量。
        """
        self.capacity = capacity
        self.cache = {}  # 哈希表：存储 key -> Node 的映射，用于O(1)查找
        self.size = 0    # 当前缓存中的元素数量

        # 使用伪头节点 (dummy head) 和伪尾节点 (dummy tail) 来简化双向链表的操作。
        # 这样可以避免在处理空链表、添加第一个节点或删除最后一个节点时的特殊情况。
        # head.next 永远指向MRU节点 (如果有的话)
        # tail.prev 永远指向LRU节点 (如果有的话)
        self.head = Node(0, 0) # 伪头节点，不存储实际数据
        self.tail = Node(0, 0) # 伪尾节点，不存储实际数据
        self.head.next = self.tail
        self.tail.prev = self.head

    def _add_node(self, node: Node):
        """
        将一个节点添加到双向链表的头部 (MRU端)。
        新节点会插入在 head 和 head.next 之间。
        """
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
        self.size += 1

    def _remove_node(self, node: Node):
        """
        从双向链表中移除一个节点。
        """
        node.prev.next = node.next
        node.next.prev = node.prev
        self.size -= 1

    def _move_to_front(self, node: Node):
        """
        将一个已存在的节点移动到双向链表的头部 (MRU端)。
        这通常发生在节点被访问 (get) 或更新 (put) 之后。
        """
        self._remove_node(node) # 先从当前位置移除
        self._add_node(node)    # 再添加到头部

    def get(self, key: int) -> int:
        """
        获取缓存中与给定键关联的值。
        如果键存在，则返回其值，并将其对应的节点移动到MRU端。
        如果键不存在，则返回 -1。
        """
        if key not in self.cache:
            return -1

        node = self.cache[key]
        self._move_to_front(node) # 访问后，将其标记为最近使用
        return node.value

    def put(self, key: int, value: int):
        """
        将键值对放入缓存。
        如果键已存在，则更新其值，并将其对应的节点移动到MRU端。
        如果键不存在：
            1. 创建新节点，并将其添加到MRU端。
            2. 如果缓存已满，则淘汰LRU节点 (即双向链表尾部的节点)。
        """
        if key in self.cache:
            # 键已存在：更新值并移动到MRU端
            node = self.cache[key]
            node.value = value
            self._move_to_front(node)
        else:
            # 键不存在：创建新节点
            new_node = Node(key, value)
            self.cache[key] = new_node
            self._add_node(new_node) # 添加到MRU端

            # 检查是否超出容量
            if self.size > self.capacity:
                # 缓存已满，需要淘汰LRU节点
                lru_node = self.tail.prev # LRU节点是伪尾节点的前一个节点
                self._remove_node(lru_node) # 从链表中移除
                del self.cache[lru_node.key] # 从哈希表中移除

print("\n示例 1: 基本操作")
lru_cache = LRUCache(2)
lru_cache.put(1, 1) # 缓存: {1:1}
lru_cache.put(2, 2) # 缓存: {1:1, 2:2}
print(f"get(1): {lru_cache.get(1)}") # 返回 1。现在 1 是MRU，2 是LRU。缓存: {2:2, 1:1}
lru_cache.put(3, 3) # 缓存已满，淘汰LRU (2)。缓存: {1:1, 3:3}
print(f"get(2): {lru_cache.get(2)}") # 返回 -1 (2已被淘汰)
lru_cache.put(4, 4) # 缓存已满，淘汰LRU (1)。缓存: {3:3, 4:4}

print("\n--- 单元测试 ---")

def run_test(test_name, cache_capacity, operations, expected_results):
    print(f"\nRunning Test: {test_name} (Capacity: {cache_capacity})")
    cache = LRUCache(cache_capacity)
    actual_results = []
    for op in operations:
        op_type = op[0]
        if op_type == "put":
            cache.put(op[1], op[2])
            actual_results.append(None) # put操作不返回结果
            print(f"  put({op[1]}, {op[2]})")
        elif op_type == "get":
            result = cache.get(op[1])
            actual_results.append(result)
            print(f"  get({op[1]}) -> {result}")
    
    # 过滤掉 put 操作的 None 结果，只比较 get 的结果
    filtered_actual = [res for res in actual_results if res is not None]
    filtered_expected = [res for res in expected_results if res is not None]

    assert filtered_actual == filtered_expected, \
        f"Test {test_name} Failed!\n  Expected: {filtered_expected}\n  Actual:   {filtered_actual}"
    print(f"Test {test_name} Passed!")

# --- 复杂度分析 ---

# 1. 时间复杂度 (Time Complexity):
#    - 对于 `get(key)` 操作：
#      - 哈希表查找 `key` 的时间复杂度平均为 O(1)。
#      - 双向链表的 `_remove_node` 和 `_add_node` 操作 (以及 `_move_to_front`)，
#        由于我们已经通过哈希表获得了节点的引用，所以这些操作都涉及固定数量的指针修改，时间复杂度为 O(1)。
#      因此，`get` 操作的平均时间复杂度为 **O(1)**。
#
#    - 对于 `put(key, value)` 操作：
#      - 哈希表查找 `key` 的时间复杂度平均为 O(1)。
#      - 如果键已存在，更新值后执行 `_move_to_front`，其内部操作时间复杂度为 O(1)。
#      - 如果键不存在，创建新节点，并执行 `_add_node` (O(1))。
#      - 如果需要淘汰，执行 `_remove_node` (O(1)) 并从哈希表删除 (O(1))。
#      因此，`put` 操作的平均时间复杂度为 **O(1)**。
#
#    - 最坏情况下：如果哈希表发生大量冲突，哈希表操作可能退化到 O(N)，但这是一个非常罕见且通常可通过良好哈希函数避免的情况。
#      在实际应用中，Python 的字典通常表现为 O(1)。

# 2. 空间复杂度 (Space Complexity):
#    - `LRUCache` 存储了两个主要的数据结构：
#      - 哈希表 `self.cache`：存储 `key -> Node` 的映射。每个缓存项占用一个哈希表条目。
#      - 双向链表：每个缓存项对应一个 `Node` 对象，每个 `Node` 对象存储 `key`, `value`, `prev`, `next`。
#    - 缓存的最大容量为 `capacity`。因此，哈希表和双向链表最多会存储 `capacity` 个数据项。
#    - 伪头节点和伪尾节点只占用常数空间。
#    因此，空间复杂度为 **O(capacity)**。
```
